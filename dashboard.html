مثال سريع (cURL)

curl -X POST https://smart-data-converter.onrender.com/v1/convert/csv \
  -H "X-API-Key: tf_live_YOUR_KEY" \
  -F "file=@data.csv"

مثال Python

import requests

r = requests.post(
    "https://smart-data-converter.onrender.com/v1/convert/csv",
    headers={"X-API-Key": "tf_live_YOUR_KEY"},
    files={"file": open("data.csv", "rb")},
)
print(r.json())

مثال JavaScript

const fd = new FormData();
fd.append("file", fileInput.files[0]);

const r = await fetch(

  {
    method: "POST",

    body: fd,
  }
);
const data = await r.json();
